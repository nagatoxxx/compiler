program ::= expr EOF

expr    ::= '\' ident+ '->' expr
          | atom+ (op expr)?

atom    ::= literal
          | ident
          | '(' expr ')'
          | op atom -- unary op application

ident   ::= ('_' | ALPHA) (ALPHA | DIGIT)*
literal ::= INT | FLOAT | STRING | CHAR
opchar  ::= '+' | '-' | '*' | '/' | '-' | '>' | '\'
op = opchar+
